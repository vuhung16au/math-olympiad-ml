% Appendix G: Fundamental Theorems of Integral Calculus

\begin{small}
\textit{Note: Only the statements of these theorems are given here. Please refer to the Calculus textbooks for proofs.}
\end{small}

\vspace{0.5em}

\begin{theorem}[Mean Value Theorem for Integrals]
If $f(x)$ is continuous over an interval $[a, b]$, then there is at least one point $c \in [a, b]$ such that
\[
f(c) = \frac{1}{b - a} \int_{a}^{b} f(x) \, dx.
\]
\end{theorem}

\begin{theorem}[Fundamental Theorem of Calculus, Part~1]
    (Integrals and Antiderivatives)
If $f(x)$ is continuous over an interval $[a, b]$, and the function $F(x)$ is defined by
\[
F(x) = \int_{a}^{x} f(t) \, dt,
\]
then
\[
F'(x) = f(x).
\]
\end{theorem}

\begin{theorem}[Fundamental Theorem of Calculus, Part~2]
    (The Evaluation Theorem)
If $f$ is continuous over the interval $[a, b]$ and $F(x)$ is any antiderivative of $f(x)$, then
\[
\int_{a}^{b} f(x) \, dx = F(b) - F(a).
\]
\end{theorem}
