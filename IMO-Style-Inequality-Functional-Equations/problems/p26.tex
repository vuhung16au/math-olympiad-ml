\subsubsection*{Problem 33. The Dyadic Parallelogram--Quadratic Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 33. The Dyadic Parallelogram--Quadratic Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map real numbers to real numbers, satisfy
\[
f(f(1))\leq1,
\]
and, for all $x,y\in\mathbb R$,
\[
8xy+3f(x)-3f(y)
\leq\frac{5\bigl(f(x+y)+f(x-y)\bigr)}2
\leq5f(x)+5y^2.
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{First set one variable to zero; can you extract a simpler relation before using the iterate bound?}

\end{exercisebox}
