\subsubsection*{Problem 12. IMO 1977 -- Problem 6}
\addcontentsline{toc}{subsubsection}{Problem 12. IMO 1977 -- Problem 6}

\begin{exercisebox}[medium]

Let $f$ be a function that maps positive integers to positive integers
and satisfies, for all $n\in\mathbb Z_{>0}$,
\[
f(n+1)>f(f(n)).
\]
Prove that
\[
f(n)=n\qquad(n\in\mathbb Z_{>0}).
\]

\medskip
\noindent
\textit{Source: International Mathematical Olympiad 1977 (19th IMO),
Day 2, Problem 6.}

\upsidehint{Can you compare consecutive values of $f$ by combining the condition with iteration?}

\end{exercisebox}
