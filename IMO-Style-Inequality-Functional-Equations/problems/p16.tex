\subsubsection*{Problem 1. The Additive--Multiplicative Rigidity}
\addcontentsline{toc}{subsubsection}{Problem 1. The Additive--Multiplicative Rigidity}

\begin{exercisebox}[hard]

Find all functions $f$ that map positive real numbers to positive real
numbers and satisfy, for all $x,y>0$,
\[
f(x+y)\geq f(x)f(y)\geq f(xy).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Evaluate at $1$, then compare repeated addition with repeated multiplication.}

\end{exercisebox}
