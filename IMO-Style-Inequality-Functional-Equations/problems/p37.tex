\subsubsection*{Problem 37. The Crossed Contraction Lock}
\addcontentsline{toc}{subsubsection}{Problem 37. The Crossed Contraction Lock}

\begin{exercisebox}[medium]

Determine all functions $f:(0,\infty)\to(0,\infty)$ such that
\[
f(f(x))\ge x
\]
for all $x>0$, and
\[
\boxed{f\bigl(xf(y)\bigr)+f\bigl(yf(x)\bigr)\le 2xy}
\]
for all $x,y>0$.

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Put $y=1$ and let $c=f(1)$. First obtain a linear upper bound for $f$, then iterate it against $f(f(x))\ge x$.}

\end{exercisebox}
