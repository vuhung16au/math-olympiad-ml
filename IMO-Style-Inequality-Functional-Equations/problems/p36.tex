\subsubsection*{Problem 36. The Multiplicative Mirror: Positive Recurrence}
\addcontentsline{toc}{subsubsection}{Problem 36. The Multiplicative Mirror: Positive Recurrence}

\begin{exercisebox}[hard]

Determine all functions $f:(0,\infty)\to(0,\infty)$ satisfying
\[
f(1)=1,\qquad f(f(x))=x
\]
for every $x>0$, and
\[
f(xy)+f\left(\frac yx\right)
\le \left(x+\frac1x\right)f(y)
\]
for all $x,y>0$.

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Follow the sequence $f(t^n)$ for $t>1$. Use positivity to obtain bounds, then apply the involution to those bounds.}

\end{exercisebox}
