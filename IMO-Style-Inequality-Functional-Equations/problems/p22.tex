\subsubsection*{Problem 29. The Conditioned Cubic Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 29. The Conditioned Cubic Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map real numbers to real numbers such that
$f(0)=0$, $f(1)=1$, and, for all $a,b,c\in\mathbb R$:
\begin{enumerate}
    \item If $a+b+c\geq0$, then
          $f(a^3)+f(b^3)+f(c^3)\geq3f(abc)$.
    \item If $a+b+c\leq0$, then
          $f(a^3)+f(b^3)+f(c^3)\leq3f(abc)$.
\end{enumerate}

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Try triples with a zero entry or two equal entries, and compare the cases of opposite signs.}

\end{exercisebox}
