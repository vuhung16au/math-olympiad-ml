\subsubsection*{Problem 10. The Reciprocal Pinch}
\addcontentsline{toc}{subsubsection}{Problem 10. The Reciprocal Pinch}

\begin{exercisebox}[medium]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that
\[
f(x)f\left(\frac1x\right)=1,
\qquad
f(2)+f\left(\frac12\right)\le\frac52,
\]
and, for all $x,y>0$,
\[
f(x+y)\ge f(x)+f(y),
\qquad
f(xy)\ge f(x)f(y).
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Use reciprocal symmetry at $2$ and compare the resulting values with their given sum bound.}

\end{exercisebox}
