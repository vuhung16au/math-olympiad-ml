\subsubsection*{Problem 65. Three Hidden Primes}
\addcontentsline{toc}{subsubsection}{Problem 65. Three Hidden Primes}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb Q\to[0,\infty)$ satisfying
\[
f(0)=0,\qquad f(30)=\frac1{30},
\]
and, for all rational numbers $x,y$,
\[
\boxed{
\begin{aligned}
f(x+y)&\le f(x)+f(y),\\
f(xy)&=f(x)f(y).
\end{aligned}}
\]

No continuity, monotonicity, or integer-valuedness is assumed.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Use base-$30$ expansions to bound $f$ on integers, then show that two distinct primes cannot both have values below $1$.}

\end{exercisebox}
