\subsubsection*{Problem 2. IMO 2013 -- Problem 5}
\addcontentsline{toc}{subsubsection}{Problem 2. IMO 2013 -- Problem 5}

\begin{exercisebox}[medium]

Let $f$ be a function that maps positive rational
numbers to real numbers and satisfies, for all
positive rational numbers $x,y$,
\[
\begin{aligned}
f(x)f(y)&\geq f(xy),\\
f(x+y)&\geq f(x)+f(y).
\end{aligned}
\]

Suppose there exists a rational number $a>1$
such that $f(a)=a$.

Prove that
\[
f(x)=x\quad\text{for all positive rational numbers }x.
\]

\medskip
\noindent
\textit{Source: International Mathematical Olympiad
2013 (54th IMO), Day 2, Problem 5.
The wording has been slightly adapted without
changing the mathematical content.}

\upsidehint{Iterate both inequalities at sums and products of $a$; can these bounds trap values at other rationals?}

\end{exercisebox}
