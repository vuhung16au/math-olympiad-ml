\subsubsection*{Problem 58. The Four-Step Prime Labyrinth}
\addcontentsline{toc}{subsubsection}{Problem 58. The Four-Step Prime Labyrinth}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb Q_{>0}\to\mathbb Q_{>0}$ satisfying
\[
f(xy)=f(x)f(y)
\]
for all positive rational $x,y$, such that $f(n)$ is a positive integer whenever $n$ is a positive integer, and
\[
\boxed{\qquad
f(f(f(f(x))))\ge x^4
\qquad(x\in\mathbb Q_{>0}).
\qquad}
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Follow prime divisors through four applications of $f$, then restart the same chain at its second prime.}

\end{exercisebox}
