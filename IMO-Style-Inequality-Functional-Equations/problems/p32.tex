\subsubsection*{Problem 5. The Triadic Balance}
\addcontentsline{toc}{subsubsection}{Problem 5. The Triadic Balance}

\begin{exercisebox}[medium]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that
\[
f(3)+f\left(\frac13\right)\le \frac{10}{3},
\]
and, for all $x,y>0$,
\[
f(x+y)\ge f(x)+f(y),
\qquad
f(xy)\le f(x)f(y).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Use repeated addition and multiplication at $3$ and $1/3$ to exploit the normalization bound.}

\end{exercisebox}
