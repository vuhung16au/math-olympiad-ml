\subsubsection*{Problem 32. The Skew-Symmetric Parallelogram Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 32. The Skew-Symmetric Parallelogram Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map real numbers to real numbers and
satisfy, for all $x,y\in\mathbb R$,
\[
2xy+x^2f(y)-y^2f(x)
\leq\frac{f(x+y)+f(x-y)}{2}
\leq f(x)+y^2.
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Swap $x$ and $y$ in the lower bound and compare the result with the upper bound.}

\end{exercisebox}
