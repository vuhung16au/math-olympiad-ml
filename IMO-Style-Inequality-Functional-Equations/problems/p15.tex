\subsubsection*{Problem 18. The Geometric-Mean Trap}
\addcontentsline{toc}{subsubsection}{Problem 18. The Geometric-Mean Trap}

\begin{exercisebox}[hard]

Find all functions $f$ of a single variable that map positive real
numbers to positive real numbers and satisfy the following property:
for every $x>0$, there exists exactly one $y>0$ such that
\[
\frac{x}{f(y)}+\frac{y}{f(x)}
\leq\frac{f(x)}x+\frac{f(y)}y.
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{For fixed $x$, rewrite the condition using the ratios $f(x)/x$ and $f(y)/y$; what does uniqueness imply?}

\end{exercisebox}
