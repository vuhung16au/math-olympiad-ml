\subsubsection*{Problem 24. The Quadratic Mean Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 24. The Quadratic Mean Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map positive real numbers to positive real
numbers, satisfy $f(f(a))\leq a^4$ for some $a>0$, and, for all
$x,y>0$,
\[
\sqrt{\frac{x^4+f(y)^2}{2}}
\geq\frac{f(x)+y^2}{2}
\geq x\sqrt{f(y)}.
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Look for choices of $x$ and $y$ that make the two mean bounds especially tight.}

\end{exercisebox}
