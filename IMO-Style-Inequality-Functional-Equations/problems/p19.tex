\subsubsection*{Problem 27. IMO 2011 -- Problem 3}
\addcontentsline{toc}{subsubsection}{Problem 27. IMO 2011 -- Problem 3}

\begin{exercisebox}[hard]

Let $f$ be a function that maps real numbers to real numbers and
satisfies, for all $x,y\in\mathbb R$,
\[
f(x+y)\leq yf(x)+f(f(x)).
\]
Prove that
\[
f(x)=0\qquad\text{for all }x\leq0.
\]

\medskip
\noindent
\textit{Source: International Mathematical Olympiad 2011 (52nd IMO),
Day 1, Problem 3.}

\upsidehint{Begin with $y=0$ and $y=-x$; what do these say about the values at $0$ and $f(x)$?}

\end{exercisebox}
