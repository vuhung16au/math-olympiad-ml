\subsubsection*{Problem 14. IMO 2023 Shortlist -- Algebra A4}
\addcontentsline{toc}{subsubsection}{Problem 14. IMO 2023 Shortlist -- Algebra A4}

\begin{exercisebox}[medium]

Find all functions $f$ of a single variable that map
positive real numbers to positive real numbers such that,
for all $x,y>0$,
\[
 x\bigl(f(x)+f(y)\bigr)
 \ge \bigl(f(f(x))+y\bigr)f(y).
\]

\noindent
\textit{Source: This problem is taken from the
IMO 2023 Shortlist, Algebra Problem A4.
The wording has been slightly adapted without
changing the mathematical content.}

\upsidehint{Fix $x$ and let $y$ vary; what does the inequality force about the possible size of $f(y)$?}

\end{exercisebox}
