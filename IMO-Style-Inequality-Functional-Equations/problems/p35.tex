\subsubsection*{Problem 35. The Subadditive--Supermultiplicative Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 35. The Subadditive--Supermultiplicative Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map positive real numbers to positive real
numbers, satisfy $f(f(a))=a$ for some $a>1$, and, for all $x,y>0$,
\[
\boxed{
f(x)\leq f(x+y)\leq f(x)+f(y)
\leq f(xy)\left(\frac1{f(x)}+\frac1{f(y)}\right).
}
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Extract monotonicity, subadditivity, and supermultiplicativity; then use the anchor to squeeze $f(x)$ against $x$.}

\end{exercisebox}
