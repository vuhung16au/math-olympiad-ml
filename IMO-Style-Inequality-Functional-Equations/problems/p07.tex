\subsubsection*{Problem 26. IMO 2023 Shortlist -- Algebra A2}
\addcontentsline{toc}{subsubsection}{Problem 26. IMO 2023 Shortlist -- Algebra A2}

\begin{exercisebox}[medium]

Let $f$ be a function that maps real numbers
to real numbers such that, for all $x,y\in\mathbb R$,
\[
f(x+y)f(x-y)\geq f(x)^2-f(y)^2.
\]

Suppose that the inequality is strict for
at least one pair of real numbers $(x_0,y_0)$.

Prove that either
\[
f(x)\geq0\quad\text{for all }x\in\mathbb R
\]
or
\[
f(x)\leq0\quad\text{for all }x\in\mathbb R.
\]

\medskip
\noindent
\textit{Source: IMO 2023 Shortlist, Algebra Problem A2,
proposed by Malaysia. The wording has been slightly
adapted without changing the mathematical content.}

\upsidehint{Compare the inequality with the one obtained by swapping $x$ and $y$.}

\end{exercisebox}
