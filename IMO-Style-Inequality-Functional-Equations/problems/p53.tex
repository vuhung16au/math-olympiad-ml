\subsubsection*{Problem 53. When Even Numbers Become Smaller}
\addcontentsline{toc}{subsubsection}{Problem 53. When Even Numbers Become Smaller}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb Q\to[0,\infty)$ satisfying
\[
f(0)=0,\qquad f(2)=\frac12,
\]
and, for all rational numbers $x,y$,
\[
\boxed{
\begin{aligned}
f(xy)&=f(x)f(y),\\
f(x+y)&\le f(x)+f(y).
\end{aligned}}
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Use binary expansions to bound $f(n)$ uniformly on positive integers, then apply the bound to $n^r$.}

\end{exercisebox}
