\subsubsection*{Problem 61. The Hidden Entropy Function}
\addcontentsline{toc}{subsubsection}{Problem 61. The Hidden Entropy Function}

\begin{exercisebox}[hard]

Determine all functions $f:(0,\infty)\to[0,\infty)$ satisfying
\[
\boxed{\qquad
f(xy)\ge x f(y)+y f(x)+(x-1)(y-1)
\qquad}
\]
for all positive real numbers $x,y$.

No continuity, monotonicity, or normalisation is assumed.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Investigate $g(x)=\bigl(f(x)+x-1\bigr)/x$, and split $x$ into arbitrarily many equal multiplicative factors.}

\end{exercisebox}
