\subsubsection*{Problem 11. The Harmonic--Product Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 11. The Harmonic--Product Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map positive real numbers to positive real
numbers, provided that $f(f(c))\geq c$ for some $c\in(0,1)$, and, for
all $x,y>0$,
\[
f\!\left(\frac{xy}{x+y}\right)
\leq\frac{f(x)f(y)}{f(x)+f(y)},
\qquad
f(xy)\geq f(x)f(y).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Try reciprocal substitutions, then bring the special point $c$ into play.}

\end{exercisebox}
