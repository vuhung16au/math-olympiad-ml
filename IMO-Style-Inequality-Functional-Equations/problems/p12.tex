\subsubsection*{Problem 23. The Euclidean--Multiplicative Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 23. The Euclidean--Multiplicative Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map positive real numbers to positive real
numbers, satisfy
\[
f(f(\sqrt2))\leq\sqrt2,
\]
and, for all $x,y>0$,
\[
f\!\left(\sqrt{x^2+y^2}\right)^2
\geq f(x)^2+f(y)^2
\geq f(xy)^2\left(\frac1{f(x)^2}+\frac1{f(y)^2}\right).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Can you choose inputs so that the Euclidean combination and the product are closely related?}

\end{exercisebox}
