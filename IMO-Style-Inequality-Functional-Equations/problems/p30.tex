\subsubsection*{Problem 20. The Cross-Dilated Shift}
\addcontentsline{toc}{subsubsection}{Problem 20. The Cross-Dilated Shift}

\begin{exercisebox}[hard]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that, for all
$x,y>0$,
\[
x^2+f\bigl(xf(y)\bigr)\leq x f(x+y).
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Compare the inequality at exchanged or rescaled inputs; what role does the sum $x+y$ play?}

\end{exercisebox}
