\subsubsection*{Problem 6. The Sum--Product Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 6. The Sum--Product Squeeze}

\begin{exercisebox}[medium]

Find all functions $f$ of a single variable that map
positive real numbers to positive real numbers and
satisfy, for all $x,y>0$,
\[
f(x+y)\bigl(f(x)+f(y)\bigr)
\le 2f(2)f(x)f(y)
\le f(xy).
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{What do the inequalities become for carefully chosen pairs such as $(x,1)$ or $(x,x)$?}

\end{exercisebox}
