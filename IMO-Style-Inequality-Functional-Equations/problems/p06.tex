\subsubsection*{Problem 25. Vietnam Mathematical Olympiad 1991}
\addcontentsline{toc}{subsubsection}{Problem 25. Vietnam Mathematical Olympiad 1991}

\begin{exercisebox}[medium]

Find all functions $f$ of a single variable that map
real numbers to real numbers and satisfy, for all
real numbers $x,y,z$,
\[
\frac{f(xy)+f(xz)}{2}
-f(x)f(yz)\geq\frac14.
\]

\noindent
\textit{Source: 29th Vietnamese Mathematical Olympiad,
1991, First Day, Problem 1.}

\upsidehint{What do the substitutions $y=z$ and $y=0$ reveal?}

\end{exercisebox}
