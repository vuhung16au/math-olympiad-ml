\subsubsection*{Problem 15. FEOO 2020 Shortlist -- Algebra A1}
\addcontentsline{toc}{subsubsection}{Problem 15. FEOO 2020 Shortlist -- Algebra A1}

\begin{exercisebox}[medium]

Find all functions $f$ of a single variable that
map real numbers to real numbers and satisfy,
for all $x,y\in\mathbb R$,
\[
2xyf(xy)\geq x^2y^2+xf(x)f(y^2).
\]

\medskip
\noindent
\textit{Source: Functional Equations Online Olympiad
(FEOO) 2020, Shortlist, Algebra Problem A1.
The wording has been slightly adapted without
changing the mathematical content.}

\upsidehint{Start with $y=0$ and exploit the fact that $y^2$ stays nonnegative.}

\end{exercisebox}
