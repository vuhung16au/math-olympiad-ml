\subsubsection*{Problem 51. The Logarithm Hidden in a Chain}
\addcontentsline{toc}{subsubsection}{Problem 51. The Logarithm Hidden in a Chain}

\begin{exercisebox}[medium]

Determine all functions $f:(0,\infty)\to\mathbb R$ satisfying
\[
\boxed{\qquad
f(xy)\le f(x)+f(y)\le x+y-2
\qquad}
\]
for all positive real numbers $x,y$.

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{First find $f(1)$, then apply the first inequality repeatedly to $n$ identical factors $x^{1/n}$.}

\end{exercisebox}
