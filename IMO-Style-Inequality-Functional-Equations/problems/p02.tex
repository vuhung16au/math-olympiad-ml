\subsubsection*{Problem 22. The Hidden Quadratic Function}
\addcontentsline{toc}{subsubsection}{Problem 22. The Hidden Quadratic Function}

\begin{exercisebox}[medium]

Find all functions $f$ of a single variable that map
positive real numbers to positive real numbers such that
$f(f(2))\le 16$ and, for all $x,y>0$,
\[
\sqrt{f(x+y)}
\ge \sqrt{f(x)}+\sqrt{f(y)}
\ge \sqrt{f(xy)}
\left(
\frac{1}{\sqrt{f(x)}}+\frac{1}{\sqrt{f(y)}}
\right).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Try setting $x=y$ and compare the resulting bounds as the input is doubled.}

\end{exercisebox}
