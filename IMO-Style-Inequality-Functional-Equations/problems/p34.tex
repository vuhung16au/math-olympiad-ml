\subsubsection*{Problem 34. The Ratio--Product Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 34. The Ratio--Product Squeeze}

\begin{exercisebox}[hard]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that, for all
$x,y>0$,
\[
x f(x+y)\ge (x+y)f(x),
\qquad
f(xy)\le f(x)f(y),
\]
and for which there exists $a>1$ satisfying
\[
f(f(a))\le f(a).
\]

\begin{flushright}
\textit{Original problem for this collection; not an official IMO problem.}
\end{flushright}

\upsidehint{Normalize by $x$ to reveal monotonicity and submultiplicativity, then propagate equality from an interval around $1$ using integer roots.}

\end{exercisebox}
