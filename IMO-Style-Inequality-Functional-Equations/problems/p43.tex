\subsubsection*{Problem 43. The Moving Zero}
\addcontentsline{toc}{subsubsection}{Problem 43. The Moving Zero}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb R\to[0,\infty)$ satisfying
\[
\boxed{\quad
f(x+y)+f(x-y)\ge 2f(x)+2y^2
\quad} \tag{1}
\]
for all real $x,y$, and
\[
\boxed{\quad f(f(x))\le x^4\quad} \tag{2}
\]
for every real $x$.

No continuity or monotonicity is assumed.

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Locate a zero $a$, repeatedly bisect the segment from $a$ to $t$, and use the resulting lower bound inside the composition inequality.}

\end{exercisebox}
