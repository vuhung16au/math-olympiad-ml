\subsubsection*{Problem 8. The Multiplicative Mirror}
\addcontentsline{toc}{subsubsection}{Problem 8. The Multiplicative Mirror}

\begin{exercisebox}[medium]

Find all functions $f$ of a single variable that map
positive real numbers to positive real numbers such that
\[
f(x)f\left(\frac1x\right)=1,
\qquad f(f(2))\le 2,
\]
and, for all $x,y>0$,
\[
\frac{f(x)f(y)}
{f(x+y)\bigl(f(x)+f(y)\bigr)}
\ge 1 \ge
\frac{f(x)^2}{f(xy)f(x/y)}.
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{What do the two inequalities say after replacing both variables by their reciprocals?}

\end{exercisebox}
