\subsubsection*{Problem 3. The Periodic Excess}
\addcontentsline{toc}{subsubsection}{Problem 3. The Periodic Excess}

\begin{exercisebox}[hard]

Let $a>1$ be a fixed real number. Find all functions $f$ that map
positive real numbers to positive real numbers and satisfy, for all
$x,y>0$,
\[
\begin{aligned}
f(x+y)&\geq f(x)+f(y),\\
f(xy)&\leq f(x)f(y),\\
f(x+a)&\leq f(x)+a.
\end{aligned}
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Iterate the shift by $a$ and compare it with what repeated addition tells you.}

\end{exercisebox}
