\subsubsection*{Problem 21. 2026 IMO Problem 5}
\addcontentsline{toc}{subsubsection}{Problem 21. 2026 IMO Problem 5}

\begin{exercisebox}[medium]

Determine all functions $f:\mathbb{R}_{>0}\to\mathbb{R}_{>0}$ such that,
for all $x,y>0$,
\[
\sqrt{\frac{x^2+f(y)^2}{2}}
\ge \frac{f(x)+y}{2}
\ge \sqrt{x f(y)}.
\]

\begin{flushright}
\textit{Source: 2026 IMO, Problem 5.}
\end{flushright}

\upsidehint{What can you learn by choosing $y$ in relation to $f(x)$ and comparing the two mean bounds?}

\end{exercisebox}
