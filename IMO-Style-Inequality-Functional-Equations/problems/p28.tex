\subsubsection*{Problem 19. The Reciprocal-Slope Pinch}
\addcontentsline{toc}{subsubsection}{Problem 19. The Reciprocal-Slope Pinch}

\begin{exercisebox}[hard]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that, for all
$x,y>0$,
\[
xy+f\bigl(xf(x)\bigr)\leq x f(x+y).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{For fixed $x$, compare the lower bounds at nearby inputs and watch how the term $xf(x)$ enters.}

\end{exercisebox}
