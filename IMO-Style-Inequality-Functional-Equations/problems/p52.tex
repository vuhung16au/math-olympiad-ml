\subsubsection*{Problem 52. The Almost-Cosine Sequence}
\addcontentsline{toc}{subsubsection}{Problem 52. The Almost-Cosine Sequence}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb Z\to\mathbb Z_{>0}$ satisfying $f(1)=2$ and
\[
\boxed{\qquad
\left|f(m+n)+f(m-n)-2f(m)f(n)\right|\le1
\qquad}
\]
for all integers $m,n$.

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{First prove unboundedness, then expand the sum of $f(x+y+z)$, $f(x+y-z)$, $f(x-y+z)$, and $f(x-y-z)$ in two ways.}

\end{exercisebox}
