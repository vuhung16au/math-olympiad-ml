\subsubsection*{Problem 54. The Harmonic Involution}
\addcontentsline{toc}{subsubsection}{Problem 54. The Harmonic Involution}

\begin{exercisebox}[medium]

Determine all functions $f:(0,\infty)\to(0,\infty)$ satisfying
\[
\boxed{\qquad f(f(x))=x\qquad}
\]
for every $x>0$, and
\[
\boxed{\qquad
f(x+y)\bigl(f(x)+f(y)\bigr)\le f(x)f(y)
\qquad}
\]
for all $x,y>0$.

No continuity, monotonicity, or normalisation is assumed.

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Prove $f(nx)\le f(x)/n$, then apply the same bound with input $f(nx)$ and use the involution.}

\end{exercisebox}
