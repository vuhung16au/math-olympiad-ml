\subsubsection*{Problem 45. The Conjugate-Radical Lock}
\addcontentsline{toc}{subsubsection}{Problem 45. The Conjugate-Radical Lock}

\begin{exercisebox}[medium]

Determine all functions $f:[0,\infty)\to[1,\infty)$ satisfying $f(1)=2$ and, for all $x,y\ge0$,
\[
\boxed{
f(x+y)\ge
f(x)f(y)+
\sqrt{\bigl(f(x)^2-1\bigr)\bigl(f(y)^2-1\bigr)}
}
\tag{1}
\]
and
\[
\boxed{\qquad f(2x)\le 2f(x)^2-1.\qquad} \tag{2}
\]

No continuity or monotonicity is assumed.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Investigate $g(x)=\log\!\left(f(x)+\sqrt{f(x)^2-1}\right)$, then combine superadditivity with exact doubling.}

\end{exercisebox}
