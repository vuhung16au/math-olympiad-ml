\subsubsection*{Problem 13. The Fixed-Point Lever}
\addcontentsline{toc}{subsubsection}{Problem 13. The Fixed-Point Lever}

\begin{exercisebox}[hard]

Find all functions $f$ that map positive real numbers
to positive real numbers, if there exists $a>0$ such that
$f(f(a))=a$, and for all $x,y>0$,
\[
f(x)\bigl(x+f(y)\bigr)
\geq f(f(x))\bigl(f(x)+y\bigr).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Try using the special point $a$ in each variable, where $f(f(a))=a$.}

\end{exercisebox}
