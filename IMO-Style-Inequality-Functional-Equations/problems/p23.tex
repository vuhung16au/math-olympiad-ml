\subsubsection*{Problem 30. The Quadratic--Difference Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 30. The Quadratic--Difference Squeeze}

\begin{exercisebox}[medium]

Find all functions $f$ that map real numbers to real numbers and
satisfy, for all $x,y\in\mathbb R$,
\[
f(x)+f(y)\geq2xy
\qquad\text{and}\qquad
f(x+y)+f(x-y)\leq2f(x)+2y^2.
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Use the first inequality to bound selected values, then feed those bounds into the second.}

\end{exercisebox}
