\subsubsection*{Problem 7. The Harmonic Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 7. The Harmonic Squeeze}

\begin{exercisebox}[medium]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that
\[
f(2)\ge \frac12
\]
and, for all $x,y>0$,
\[
f(x+y)\bigl(f(x)+f(y)\bigr)
\le f(x)f(y)
\le f(xy).
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Set $x=y$ first, then compare the resulting doubling relation with the product bound.}

\end{exercisebox}
