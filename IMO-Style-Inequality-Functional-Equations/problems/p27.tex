\subsubsection*{Problem 4. The Dyadic Lock}
\addcontentsline{toc}{subsubsection}{Problem 4. The Dyadic Lock}

\begin{exercisebox}[medium]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that
\[
f(2)f(f(2))\leq4
\]
and, for all $x,y>0$,
\[
f(x+y)\geq f(x)+f(y)
\geq f(xy)\left(\frac1{f(x)}+\frac1{f(y)}\right).
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{What does the inequality at equal inputs tell you, and how might it interact with the bound at $2$?}

\end{exercisebox}
