\subsubsection*{Problem 28. The Degree-Normalized Cubic Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 28. The Degree-Normalized Cubic Squeeze}

\begin{exercisebox}[medium]

Find all functions $f$ that map positive real numbers to real numbers
and satisfy, for all $x,y>0$,
\[
\frac{f(x)}{x^2}+\frac{f(y)}{y^2}
\leq\frac{f(x+y)}{(x+y)^2}
\qquad\text{and}\qquad
\frac{f(x)}{x^3}+\frac{f(y)}{y^3}
\geq\frac{2f(x+y)}{(x+y)^3}.
\]

\upsidehint{Normalize by a power of the input; which choice makes one of the inequalities simpler?}

\end{exercisebox}
