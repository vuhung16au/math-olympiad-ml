\subsubsection*{Problem 66. The Irrational Iteration Root}
\addcontentsline{toc}{subsubsection}{Problem 66. The Irrational Iteration Root}

\begin{exercisebox}[hard]

Determine all functions $f:(0,\infty)\to(0,\infty)$ satisfying
\[
\boxed{
\begin{aligned}
f(x+y)&\ge f(x)+f(y),\\
f(xy)&\le f(x)f(y)
\end{aligned}}
\]
for all positive real numbers $x,y$, together with
\[
\boxed{\qquad f(f(x))=x^2\qquad(x>0).\qquad}
\]

No continuity, monotonicity, or normalisation is assumed.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Apply $f$ to $f(xy)\le f(x)f(y)$, then apply submultiplicativity to $f(x)$ and $f(y)$.}

\end{exercisebox}
