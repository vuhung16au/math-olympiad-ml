\subsubsection*{Problem 31. The Iterated Parallelogram--Quadratic Squeeze}
\addcontentsline{toc}{subsubsection}{Problem 31. The Iterated Parallelogram--Quadratic Squeeze}

\begin{exercisebox}[hard]

Find all functions $f$ that map real numbers to real numbers, satisfy
\[
f(f(2))\leq16,
\]
and, for all $x,y\in\mathbb R$,
\[
 x^2-y^2+2yf(x)+f(y)-f(f(x))
 \leq\frac{f(x+y)+f(x-y)}{2}
 \leq f(x)+y^2.
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Start with $x=0$ and $y=0$; what bounds do they give before you iterate $f$?}

\end{exercisebox}
