\subsubsection*{Problem 44. The Hidden Golden Exponent}
\addcontentsline{toc}{subsubsection}{Problem 44. The Hidden Golden Exponent}

\begin{exercisebox}[medium]

Determine all functions $f:(0,\infty)\to(0,\infty)$ satisfying
\[
f(xy)=f(x)f(y)
\]
and
\[
\boxed{\qquad
f\bigl(x+f(y)\bigr)\ge f(x)+y f(y)
\qquad}
\]
for all positive real numbers $x,y$.

No continuity or monotonicity is assumed.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Set $y=1$, use multiplicativity to obtain superadditivity, and only then investigate power functions.}

\end{exercisebox}
