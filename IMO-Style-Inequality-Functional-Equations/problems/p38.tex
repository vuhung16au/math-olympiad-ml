\subsubsection*{Problem 38. The Injective Staircase}
\addcontentsline{toc}{subsubsection}{Problem 38. The Injective Staircase}

\begin{exercisebox}[medium]

Determine all injective functions $f:\mathbb Z_{>0}\to\mathbb Z_{>0}$ such that
\[
\boxed{\quad f\bigl(m+f(n)\bigr)\le n+f(m+1)-1\quad}
\]
for all positive integers $m,n$.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Put $n=1$ and $n=2$ to get "staircase" bounds. Then compare how many distinct values must fit under a linear bound.}

\end{exercisebox}
