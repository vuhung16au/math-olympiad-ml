\subsubsection*{Problem 17. IMO 2022 Shortlist -- Algebra A3}
\addcontentsline{toc}{subsubsection}{Problem 17. IMO 2022 Shortlist -- Algebra A3}

\begin{exercisebox}[medium]

Find all functions $f$ of a single variable that map
positive real numbers to positive real numbers such
that, for every $x>0$, there exists exactly one
$y>0$ satisfying
\[
xf(y)+yf(x)\leq2.
\]

\medskip
\noindent
\textit{Source: International Mathematical Olympiad
2022 (63rd IMO), Shortlist, Algebra Problem A3,
proposed by the Netherlands. The wording has been
slightly adapted without changing the mathematical
content.}

\upsidehint{For a fixed $x$, what shape does the set of admissible $y$ have, and how does uniqueness restrict it?}

\end{exercisebox}
