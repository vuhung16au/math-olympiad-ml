\subsubsection*{Problem 42. Bounded Errors, Exact Answer}
\addcontentsline{toc}{subsubsection}{Problem 42. Bounded Errors, Exact Answer}

\begin{exercisebox}[medium]

Determine all functions $f:\mathbb R\to\mathbb R$ satisfying $f(2)>1$ and, for all real numbers $x,y$,
\[
\boxed{
\begin{aligned}
\bigl|f(x+y)-f(x)-f(y)\bigr|&\le1,\\
\bigl|f(xy)-f(x)f(y)\bigr|&\le1.
\end{aligned}}
\]

No continuity, monotonicity, or boundedness is assumed.

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{First prove that $f$ is unbounded, then compare the approximations to $f((xy)t)$ and $f(x(yt))$.}

\end{exercisebox}
