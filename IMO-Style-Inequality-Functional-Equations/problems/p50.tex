\subsubsection*{Problem 50. The Golden Staircase}
\addcontentsline{toc}{subsubsection}{Problem 50. The Golden Staircase}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb Z_{>0}\to\mathbb Z_{>0}$ such that, for all positive integers $m,n$,
\[
\boxed{\qquad
0\le f(m+n)-f(m)-f(n)\le1
\qquad} \tag{1}
\]
and
\[
\boxed{\qquad
\left|f(f(n))-f(n)-n\right|\le1.
\qquad} \tag{2}
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Compare $f(mn)$ with both $m f(n)$ and $n f(m)$ to uncover a limiting slope.}

\end{exercisebox}
