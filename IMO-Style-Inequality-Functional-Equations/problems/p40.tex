\subsubsection*{Problem 40. The Monotonicity Ratchet}
\addcontentsline{toc}{subsubsection}{Problem 40. The Monotonicity Ratchet}

\begin{exercisebox}[medium]

Determine all functions $f:(0,\infty)\to(0,\infty)$ such that
\[
\boxed{\quad f\bigl(x+f(y)\bigr)\ge f(x)+y\qquad\text{and}\qquad f\bigl(f(y)\bigr)\le y\quad}
\]
for all $x,y>0$.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Show that $f(t)>y$ whenever $t>f(y)$, then substitute $y=f(w)$ to prove $f$ is strictly increasing.}

\end{exercisebox}
