\subsubsection*{Problem 9. The Reflected Product}
\addcontentsline{toc}{subsubsection}{Problem 9. The Reflected Product}

\begin{exercisebox}[medium]

Find all functions $f:(0,\infty)\to(0,\infty)$ such that
\[
f(x)f\left(\frac1x\right)=1,
\qquad f(2)\leq 2,
\]
and, for all $x,y>0$,
\[
f(x+y)\geq f(x)+f(y),
\qquad
f(x)^2\leq f(xy)f\left(\frac{x}{y}\right).
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Combine the reciprocal identity with the product inequality after interchanging $x$ and $y$.}

\end{exercisebox}
