\subsubsection*{Problem 68. The Conditioned Bilinear Sign Test}
\addcontentsline{toc}{subsubsection}{Problem 68. The Conditioned Bilinear Sign Test}

\begin{exercisebox}[hard]

Find all functions $f:\mathbb R\to\mathbb R$ satisfying $f(0)=0$, $f(1)=1$, and,
for all $a,b,c,d\in\mathbb R$:
\begin{enumerate}
    \item If $ab+cd\geq0$, then
          \[
          f(a)f(b)+f(c)f(d)\geq0.
          \]
    \item If $ab+cd\leq0$, then
          \[
          f(a)f(b)+f(c)f(d)\leq0.
          \]
\end{enumerate}

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026.}
\end{flushright}

\upsidehint{Use the boundary case $ab+cd=0$ to obtain exact identities. Try
$(t,1,-t,1)$ and $(x,y,-xy,1)$, then take logarithms on the positive half-line.}

\end{exercisebox}
