\subsubsection*{Problem 49. The Prime-Pairing Machine}
\addcontentsline{toc}{subsubsection}{Problem 49. The Prime-Pairing Machine}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb Q_{>0}\to\mathbb Q_{>0}$ satisfying
\[
\boxed{
f(xy)=f(x)f(y),
\qquad
f(f(x))\ge x^2
}
\]
for all positive rational numbers $x,y$, and such that
\[
\boxed{\qquad f(n)\in\mathbb Z_{>0}\quad\text{whenever }n\in\mathbb Z_{>0}.\qquad}
\]

\begin{flushright}
\textit{Proposed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Use reciprocals to force equality, then track what happens to each prime divisor of $f(p)$.}

\end{exercisebox}
