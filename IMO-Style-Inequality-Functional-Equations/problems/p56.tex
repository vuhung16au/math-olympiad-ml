\subsubsection*{Problem 56. The Microscopic Error Trap}
\addcontentsline{toc}{subsubsection}{Problem 56. The Microscopic Error Trap}

\begin{exercisebox}[medium]

Determine all functions $f:\mathbb R\to\mathbb R$ satisfying
\[
\boxed{\qquad
0\le f(x)+f(y)-f(x+y)\le(x-y)^2
\qquad} \tag{1}
\]
for all real $x,y$, and
\[
\boxed{\qquad
f\bigl(f(x^2)\bigr)\ge f(x)^2
\qquad} \tag{2}
\]
for every real $x$.

No continuity, monotonicity, or normalisation is assumed.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Use exact doubling to shrink the additive error, then study the sign of the composite function $f\circ f$.}

\end{exercisebox}
