\subsubsection*{Problem 55. The Reversible Difference}
\addcontentsline{toc}{subsubsection}{Problem 55. The Reversible Difference}

\begin{exercisebox}[medium]

Determine all functions $f:\mathbb Z\to\mathbb Z_{\ge0}$ satisfying
\[
f(0)=0,\qquad f(1)=1,\qquad f(f(2))\le16,
\]
and, for all integers $m,n$,
\[
\boxed{\qquad
f(m+3n)+3f(m+n)
\ge 3f(m+2n)+f(m).
\qquad}
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Replace $(m,n)$ by $(m+3n,-n)$, then examine consecutive second differences.}

\end{exercisebox}
