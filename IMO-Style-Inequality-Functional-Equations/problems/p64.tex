\subsubsection*{Problem 64. The Invisible Conjugate}
\addcontentsline{toc}{subsubsection}{Problem 64. The Invisible Conjugate}

\begin{exercisebox}[hard]

Let
\[
K=\{a+b\sqrt2:a,b\in\mathbb Q\}.
\]
Determine all functions $f:K\to[0,\infty)$ satisfying
\[
f(0)=0,\qquad f(1)=1,\qquad f(2)=2,
\]
and, for all $x,y\in K$,
\[
\boxed{
\begin{aligned}
f(x+y)&\le f(x)+f(y),\\
f(xy)&=f(x)f(y).
\end{aligned}}
\]

No continuity, monotonicity, or symmetry is assumed.

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{For rational $r$ close to $\sqrt2$, examine the product $f(\sqrt2-r)f(\sqrt2+r)$.}

\end{exercisebox}
