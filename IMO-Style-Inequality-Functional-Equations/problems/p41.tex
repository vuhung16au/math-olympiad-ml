\subsubsection*{Problem 41. The Prime-Exponent Trap}
\addcontentsline{toc}{subsubsection}{Problem 41. The Prime-Exponent Trap}

\begin{exercisebox}[hard]

Determine all functions $f:\mathbb Q_{>0}\to\mathbb Q_{>0}$ such that, for all positive rational numbers $x,y$,
\[
f(xy)=f(x)f(y)
\]
and
\[
\boxed{\qquad f\bigl(f(x)^3\bigr)\ge x^2f(x).\qquad}
\]

\begin{flushright}
\textit{Composed by Vu Hung Nguyen, 2026}
\end{flushright}

\upsidehint{Replace $x$ by $1/x$, then investigate how prime exponents behave along the iterates of $f(x)/x$.}

\end{exercisebox}
